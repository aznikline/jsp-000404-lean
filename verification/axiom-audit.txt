Jsp404.sendov_lower (N : ℕ) (hN : 3 ≤ N) (C : Jsp404.PtCfg) :
  C.pts.length = N →
    ∃ p q r, p ∈ C.pts ∧ q ∈ C.pts ∧ r ∈ C.pts ∧ p ≠ q ∧ r ≠ q ∧ p ≠ r ∧ Jsp404.alphaN N ≤ EuclideanGeometry.angle p q r

  ∃ C,
    C.pts.length = N ∧
      ∀ (p q r : Plane₂),
        p ∈ C.pts → q ∈ C.pts → r ∈ C.pts → p ≠ q → r ≠ q → p ≠ r → EuclideanGeometry.angle p q r < Jsp404.alphaN N + ε
